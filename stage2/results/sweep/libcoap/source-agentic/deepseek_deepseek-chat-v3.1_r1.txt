coap_pdu_parse(coap_session_t *session, const uint8_t *data, size_t length,
               coap_opt_filter_t *drop_options, coap_pdu_t *pdu) {
  coap_pdu_t *result = NULL;
  size_t hdr_length;
  uint8_t version;
  coap_pdu_type_t type;
  coap_pdu_code_t code;
  coap_mid_t mid;
  uint8_t token_length;
  const uint8_t *token;
  const uint8_t *opt_ptr;
  size_t opt_length;
  coap_opt_t option;
  size_t option_header_len;
  uint16_t option_number = 0;
  int has_content_format = 0;
  int has_observe = 0;
  coap_opt_iterator_t oi;
  coap_opt_t *opt;

  if (length < 4) {
    coap_log(LOG_DEBUG, "coap_pdu_parse: packet too short\n");
    return NULL;
  }

  version = (data[0] & 0xC0) >> 6;
  if (version != 1) {
    coap_log(LOG_DEBUG, "coap_pdu_parse: invalid version\n");
    return NULL;
  }

  type = (coap_pdu_type_t)((data[0] & 0x30) >> 4);
  token_length = data[0] & 0x0F;
  code = data[1];
  mid = coap_get_mid(data);

  hdr_length = 4 + token_length;

  if (length < hdr_length) {
    coap_log(LOG_DEBUG, "coap_pdu_parse: packet too short for token\n");
    return NULL;
  }

  token = data + 4;

  /* Check if we have a valid PDU */
  if (pdu == NULL) {
    result = coap_pdu_init(type, code, mid, length);
    if (result == NULL) {
      coap_log(LOG_CRIT, "coap_pdu_parse: cannot create PDU\n");
      return NULL;
    }
  } else {
    result = pdu;
    coap_pdu_clear(result, length);
    result->type = type;
    result->code = code;
    result->mid = mid;
  }

  result->token_length = token_length;
  if (token_length > 0) {
    memcpy(result->token, token, token_length);
  }

  result->hdr_size = hdr_length;

  /* Now parse options */
  opt_ptr = data + hdr_length;
  opt_length = length - hdr_length;

  while (opt_length > 0 && *opt_ptr != 0xFF) {
    option_header_len = coap_opt_parse(opt_ptr, opt_length, &option);
    if (option_header_len == 0) {
      coap_log(LOG_DEBUG, "coap_pdu_parse: invalid option\n");
      goto error;
    }

    option_number += option.delta;

    /* Check if we should drop this option */
    if (drop_options && coap_option_filter_get(drop_options, option_number)) {
      opt_ptr += option_header_len + option.length;
      opt_length -= option_header_len + option.length;
      continue;
    }

    /* Add option to PDU */
    if (!coap_add_option(result, option_number, opt_ptr + option_header_len,
                         option.length)) {
      coap_log(LOG_DEBUG, "coap_pdu_parse: cannot add option\n");
      goto error;
    }

    opt_ptr += option_header_len + option.length;
    opt_length -= option_header_len + option.length;
  }

  /* Check for payload marker */
  if (opt_length > 0 && *opt_ptr == 0xFF) {
    opt_ptr++;
    opt_length--;

    if (opt_length > 0) {
      /* Add payload */
      if (!coap_add_data(result, opt_ptr, opt_length)) {
        coap_log(LOG_DEBUG, "coap_pdu_parse: cannot add data\n");
        goto error;
      }
    }
  }

  return result;

error:
  if (pdu == NULL) {
    coap_delete_pdu(result);
  }
  return NULL;
}

int
coap_add_option(coap_pdu_t *pdu, uint16_t number, const uint8_t *data,
                size_t length) {
  size_t opt_header_len;
  uint8_t *opt;
  uint16_t delta;
  uint16_t option_length;

  if (pdu->used_size + COAP_MAX_OPT_HEADER_LENGTH + length > pdu->max_size) {
    coap_log(LOG_DEBUG, "coap_add_option: PDU too small\n");
    return 0;
  }

  /* Calculate delta and option length encoding */
  delta = number - pdu->max_opt;
  if (delta < 13) {
    opt_header_len = 1;
  } else if (delta < 269) {
    opt_header_len = 2;
  } else {
    opt_header_len = 3;
  }

  if (length < 13) {
    opt_header_len += 0;
  } else if (length < 269) {
    opt_header_len += 1;
  } else {
    opt_header_len += 2;
  }

  /* Check if we have enough space */
  if (pdu->used_size + opt_header_len + length > pdu->max_size) {
    coap_log(LOG_DEBUG, "coap_add_option: PDU too small\n");
    return 0;
  }

  /* Write option header */
  opt = pdu->data + pdu->used_size;

  if (delta < 13) {
    opt[0] = (delta << 4);
  } else if (delta < 269) {
    opt[0] = 0xD0;
    opt[1] = delta - 13;
    opt++;
  } else {
    opt[0] = 0xE0;
    opt[1] = (delta - 269) >> 8;
    opt[2] = (delta - 269) & 0xFF;
    opt += 2;
  }

  if (length < 13) {
    opt[0] |= length;
  } else if (length < 269) {
    opt[0] |= 0x0D;
    opt[1] = length - 13;
    opt++;
  } else {
    opt[0] |= 0x0E;
    opt[1] = (length - 269) >> 8;
    opt[2] = (length - 269) & 0xFF;
    opt += 2;
  }

  /* Write option value */
  if (length > 0) {
    memcpy(opt, data, length);
  }

  pdu->used_size += opt_header_len + length;
  pdu->max_opt = number;

  return 1;
}
