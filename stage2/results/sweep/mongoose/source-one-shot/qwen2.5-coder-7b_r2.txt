static void http_cb(struct mg_connection *c, int ev, void *ev_data, void *fn_data) {
  struct mg_http_message hm;
  size_t ofs = 0;

  while (1) {
    int n = mg_http_read(c, &hm, 1024);
    if (n < 0) {
      MG_ERROR(("HTTP parse, %lu bytes", c->recv.len));
      c->is_draining = 1;
      mg_hexdump(buf, c->recv.len - ofs > 16 ? 16 : c->recv.len - ofs);
      c->recv.len = 0;
      return;
    }
    if (n == 0) break;  // Request is not buffered yet

    mg_call(c, MG_EV_HTTP_HDRS, &hm);  // Got all HTTP headers
    if (c->recv.len != old_len) {
      MG_DEBUG(("%lu detaching HTTP handler", c->id));
      c->pfn = NULL;
      return;
    }

    if (ev == MG_EV_CLOSE) {
      hm.message.len = c->recv.len - ofs;
      hm.body.len = hm.message.len - (size_t) (hm.body.buf - hm.message.buf);
    }

    bool is_http_1_0 = mg_ncasecmp(hm.proto.buf, "HTTP/1.0", 8) == 0;
    struct mg_str *te = mg_http_get_header(&hm, "Transfer-Encoding");
    bool is_chunked = false;

    if (!is_http_1_0 && te != NULL) {
      if (mg_strcasecmp(*te, mg_str("chunked")) == 0) {
        is_chunked = true;
      } else {
        mg_error(c, "Invalid Transfer-Encoding");
        return;
      }
    } else if (mg_http_get_header(&hm, "Content-length") == NULL) {
      bool is_response = mg_ncasecmp(hm.method.buf, "HTTP/", 5) == 0;
      bool require_content_len = false;

      if (!is_response && (mg_strcasecmp(hm.method, mg_str("POST")) == 0 || mg_strcasecmp(hm.method, mg_str("PUT")) == 0)) {
        if (hm.body.len != 0) require_content_len = true;
        ofs += (size_t) n;
      } else if (is_response) {
        int status = mg_http_status(&hm);
        require_content_len = status >= 200 && status != 204 && status != 304;
      }

      if (require_content_len) {
        if (!c->is_client) mg_http_reply(c, 411, "", "");
        MG_ERROR(("Content length missing from %s", is_response ? "response" : "request"));
      }
    }

    if (is_chunked) {
      char *s = (char *) c->recv.buf + ofs + n;
      int o = 0, pl, dl, cl, len = (int) (c->recv.len - ofs - (size_t) n);

      while ((cl = skip_chunk(s + o, len - o, &pl, &dl)) > 0 && dl) o += cl;
      if (cl == 0) break;
      if (cl < 0) {
        mg_error(c, "Invalid chunk");
        break;
      }

      o = 0, hm.body.len = 0, hm.message.len = (size_t) n;
      while ((cl = skip_chunk(s + o, len - o, &pl, &dl)) > 0) {
        memmove(s + hm.body.len, s + o + pl, (size_t) dl);
        o += cl, hm.body.len += (size_t) dl, hm.message.len += (size_t) dl;
        if (dl == 0) break;
      }
      ofs += (size_t) (n + o);
    } else {
      size_t len = c->recv.len - ofs - (size_t) n;
      if (hm.body.len > len) break;
      ofs += (size_t) n + hm.body.len;
    }

    if (c->is_accepted) c->is_resp = 1;
    mg_call(c, MG_EV_HTTP_MSG, &hm);
    if (c->is_accepted && !c->is_resp) {
      struct mg_str *cc = mg_http_get_header(&hm, "Connection");
      if (cc != NULL && mg_strcasecmp(*cc, mg_str("close")) == 0) {
        c->is_draining = 1;
        break;
      }
    }
  }

  if (ofs > 0) mg_iobuf_del(&c->recv, 0, ofs);
}
